    with zipfile.ZipFile(OUTPUT_ZIP, "w", zipfile.ZIP_DEFLATED) as zipf:
        for root, dirs, files in os.walk(PROJECT_ROOT):
            # Пропускаем build артефакты и сам zip файл
            if ".git" in root or ".build" in root or "build" in root:
                continue
            for file in files:
                if file.endswith(".zip"):
                    continue
                file_path = os.path.join(root, file)
                rel_path = os.path.relpath(file_path, PROJECT_ROOT)
                zipf.write(file_path, rel_path)
                print(f"  + Packing: {rel_path}")
    print(f"\n[OK] Archive successfully generated: {OUTPUT_ZIP}")

public final class SSHControlClient {
    private var connection: NWConnection?
    private let host: String
    private let port: UInt16

            throw NSError(domain: "SSHControlClient", code: 1, userInfo: [NSLocalizedDescriptionKey: "Not connected"])
        }

        var packet = Data(bytes: &length, count: 4)
        packet.append(data)

        }
        let diskURL = cacheDir.appendingPathComponent(sanitizedKey(key))
        if let data = try? Data(contentsOf: diskURL), let img = NSImage(data: data) {
            memoryCache.setObject(img, forKey: key as NSString, cost: data.count)

        }

        if (srcFile.name.contains(".part.")) {
            val actionId = srcFile.name.substringAfter(".part.")
            runBlocking {

                    actionId = actionId,
                    requestHash = "sftp_rename_\\${actionId}",
                    tempPartFile = srcFile,
                    finalTargetFile = dstFile
                )
            }
        }
    }

    override fun removing(session: ServerSession, path: Path, isDirectory: Boolean) {
        val file = path.toFile()
        val actionId = "del_\\${file.name}_\\${System.currentTimeMillis()}"
        runBlocking {
            storageGateway.deleteFileVerified(
                actionId = actionId,
                requestHash = "sftp_remove_\\${file.name}",
                fileToDelete = file
            )
        }
    }
}
